320void CIPHERAES::AESGenTables(void)
321{
322 int pow[256];
323 int log[256];
324 int i;
325 int x;
326 int y;
327 int z;
328
329 // Compute pow and log tables over GF(2^8)
330
331 for(i=0, x=1; i<256; i++)
332 {
333 pow[i] = x;
334 log[x] = i;
335 x = ( x ^ XTIME( x ) ) & 0xFF;
336 }
337
338
339 // Calculate the round constants
340
341 for(i=0, x=1; i<10; i++)
342 {
343 RCON[i] = (XDWORD) x;
344 x = XTIME( x ) & 0xFF;
345 }
346
347
348 // Generate the forward and reverse S-boxes
349
350 FSb[0x00] = 0x63;
351 RSb[0x63] = 0x00;
352
353 for(i=1; i<256; i++)
354 {
355 x = pow[255 - log[i]];
356
357 y = x; y = ( (y << 1) | (y >> 7) ) & 0xFF;
358 x ^= y; y = ( (y << 1) | (y >> 7) ) & 0xFF;
359 x ^= y; y = ( (y << 1) | (y >> 7) ) & 0xFF;
360 x ^= y; y = ( (y << 1) | (y >> 7) ) & 0xFF;
361 x ^= y ^ 0x63;
362
363 FSb[i] = (XBYTE) x;
364 RSb[x] = (XBYTE) i;
365 }
366
367
368 // Generate the forward and reverse tables
369
370 for( i = 0; i < 256; i++ )
371 {
372 x = FSb[i];
373 y = XTIME( x ) & 0xFF;
374 z = ( y ^ x ) & 0xFF;
375
376 FT0[i] = ( (XDWORD) y ) ^
377 ( (XDWORD) x << 8 ) ^
378 ( (XDWORD) x << 16 ) ^
379 ( (XDWORD) z << 24 );
380
381 FT1[i] = ROTL8( FT0[i] );
382 FT2[i] = ROTL8( FT1[i] );
383 FT3[i] = ROTL8( FT2[i] );
384
385 x = RSb[i];
386
387 RT0[i] = ( (XDWORD) MUL( 0x0E, x ) ) ^
388 ( (XDWORD) MUL( 0x09, x ) << 8 ) ^
389 ( (XDWORD) MUL( 0x0D, x ) << 16 ) ^
390 ( (XDWORD) MUL( 0x0B, x ) << 24 );
391
392 RT1[i] = ROTL8( RT0[i] );
393 RT2[i] = ROTL8( RT1[i] );
394 RT3[i] = ROTL8( RT2[i] );
395 }
396}

412bool CIPHERAES::AESSetKeyCipher(CIPHERAES_CONTEXT* ctx, XBYTE* key, XDWORD keysize)
413{
414 XDWORD* RK;
415
416 switch( keysize )
417 {
418 case 128 : ctx->nr = 10; break;
419 case 192 : ctx->nr = 12; break;
420 case 256 : ctx->nr = 14; break;
421 default : return false;
422 }
423
424 ctx->rk = RK = ctx->buf;
425
426 for(XDWORD i = 0; i < (keysize >> 5); i++ )
427 {
428 GET_UINT32_LE( RK[i], key, i << 2 );
429 }
430
431 switch( ctx->nr )
432 {
433 case 10: for(XDWORD i = 0; i < 10; i++, RK += 4 )
434 {
435 RK[4] = RK[0] ^ RCON[i] ^
436 ( (XDWORD) FSb[ ( RK[3] >> 8 ) & 0xFF ] ) ^
437 ( (XDWORD) FSb[ ( RK[3] >> 16 ) & 0xFF ] << 8 ) ^
438 ( (XDWORD) FSb[ ( RK[3] >> 24 ) & 0xFF ] << 16 ) ^
439 ( (XDWORD) FSb[ ( RK[3] ) & 0xFF ] << 24 );
440
441 RK[5] = RK[1] ^ RK[4];
442 RK[6] = RK[2] ^ RK[5];
443 RK[7] = RK[3] ^ RK[6];
444 }
445 break;
446
447 case 12: for(XDWORD i = 0; i < 8; i++, RK += 6 )
448 {
449 RK[6] = RK[0] ^ RCON[i] ^
450 ( (XDWORD) FSb[ ( RK[5] >> 8 ) & 0xFF ] ) ^
451 ( (XDWORD) FSb[ ( RK[5] >> 16 ) & 0xFF ] << 8 ) ^
