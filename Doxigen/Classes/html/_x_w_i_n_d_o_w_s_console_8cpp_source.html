624void XWINDOWSCONSOLE::SendEnterKey()
625{
626 #ifndef APPFLOW_GRAPHICS_NOTCONSOLE_ACTIVE
627
628 INPUT ip;
629 // Set up a generic keyboard event.
630 ip.type = INPUT_KEYBOARD;
631 ip.ki.wScan = 0; // hardware scan code for key
632 ip.ki.time = 0;
633 ip.ki.dwExtraInfo = 0;
634
635 // Send the "Enter" key
636 ip.ki.wVk = 0x0D; // virtual-key code for the "Enter" key
637 ip.ki.dwFlags = 0; // 0 for key press
638 SendInput(1, &ip, sizeof(INPUT));
639
640 // Release the "Enter" key
641 ip.ki.dwFlags = KEYEVENTF_KEYUP; // KEYEVENTF_KEYUP for key release
642 SendInput(1, &ip, sizeof(INPUT));
643
644 #endif
645}
