195bool XLICENSE::GenerateMachineID(XLICENSEID& licenseID)
196{
197 XSTRING origin;
198 XBUFFER originbuffer;
199 HASHSHA2* sha2 = NULL;
200 XUUID ID;
201 XBUFFER* result = NULL;
202
203 ID.Empty();
204
205 //GEN_XSYSTEM.GetOperativeSystemID(origin);
206
207 origin.AddFormat(__L(" %s"), GEN_XSYSTEM.GetBIOSSerialNumber()->Get());
208 origin.AddFormat(__L(" %s"), GEN_XSYSTEM.GetCPUSerialNumber()->Get());
209
210 origin.ConvertToUTF8(originbuffer);
211
212 sha2 = new HASHSHA2(HASHSHA2TYPE_256);
213 if(!sha2)
214 {
215 return false;
216 }
217
218 sha2->Do(originbuffer);
219 result = sha2->GetResult();
220
221 if(!result)
222 {
223 return false;
224 }
225
226 if(!result->GetSize())
227 {
228 return false;
229 }
230
231 XDWORD data1 = 0;
232 XWORD data2 = 0;
233 XWORD data3 = 0;
234 XBYTE data4 = 0;
235 XBYTE data5 = 0;
236 XBYTE* data6 = &result->Get()[12];
237
238 result->Get((XDWORD&)data1);
239 result->Get((XWORD&)data2);
240 result->Get((XWORD&)data3);
241 result->Get((XBYTE&)data4);
242 result->Get((XBYTE&)data5);
243
244 ID.Set(data1, data2, data3, data4, data5, data6);
245
246 ID.GetToString(origin);
247
248 delete sha2;
249
250 licenseID.GetID()->Set(ID);
251
252 return true;
253}

269bool XLICENSE::Generate(XLICENSEID& licenseID, XSTRING* applicationID, XBUFFER* license)
270{
271 CIPHERAES cipherAES;
272 CIPHERKEYSYMMETRICAL key;
273 XBUFFER xbufferkey;
274 XBUFFER xbuffercipher;
275 XBUFFER* _license = license;
276 XSTRING* _applicationID = applicationID;
277
278 if(!_license) _license = this->license;
279 if(!_license)
280 {
281 return false;
282 }
283
284 if(!_applicationID) _applicationID = &this->applicationID;
285 if(!_applicationID)
286 {
287 return false;
288 }
289
290 _license->Delete();
291 licenseID.GetID()->GetToBuffer(xbuffercipher);
292
293 GetBufferKeyFromMachineID((*_applicationID), xbufferkey);
294
295 key.Set(xbufferkey);
296
297 cipherAES.SetKey(&key);
298 cipherAES.SetChainingMode(CIPHERCHAININGMODE_CBC);
299 cipherAES.SetPaddingType(XBUFFER_PADDINGTYPE_NONE);
300
301 if(cipherAES.Cipher(xbuffercipher))
302 {
303 _license->Delete();
304 _license->Add(cipherAES.GetResult());
305
306 return true;
307 }
308
309 return false;

360bool XLICENSE::CipherExpirationDate(bool cipher, XSTRING& applicationID, XSTRING& expirationdate)
361{
362 XBUFFER xbufferkey;
363 CIPHERAES cipherAES;
364 CIPHERKEYSYMMETRICAL key;
365 bool status = false;
366
367 if(!GetBufferKeyFromMachineID(applicationID, xbufferkey))
368 {
369 return false;
370 }
371
372 key.Set(xbufferkey);
373
374 cipherAES.SetKey(&key);
375 cipherAES.SetChainingMode(CIPHERCHAININGMODE_CBC);
376 cipherAES.SetPaddingType(XBUFFER_PADDINGTYPE_SPACES);
377
378 if(cipher)
379 {
380 XBUFFER charstr;
381
382 expirationdate.ConvertToASCII(charstr);
383
384 status = cipherAES.Cipher((XBYTE*)charstr.Get(), expirationdate.GetSize());
385 if(status)
386 {
387 expirationdate.Empty();
388 cipherAES.GetResult()->ConvertToBase64(expirationdate);
389 return true;
390 }
391 }
392 else
393 {
394 XBUFFER xbuffer;
395
396 xbuffer.ConvertFromBase64(expirationdate);
397
398 status = cipherAES.Uncipher(xbuffer);
399 if(status)
400 {
401 expirationdate.Empty();
402
403 for(XDWORD c=0; c<cipherAES.GetResult()->GetSize(); c++)
404 {
405 expirationdate.Add((XCHAR)cipherAES.GetResult()->GetByte(c));
406 }
407
408 expirationdate.DeleteCharacter(0x20, XSTRINGCONTEXT_TO_END);
409
410 return true;
411 }
412 }
413
414 return status;
415}

600bool XLICENSE::LoadFromBuffer(XBUFFER& xbuffer, XSTRING& applicationID, XBUFFER* license, XSTRING* expirationdate)
601{
602 if(!license)
603 {
604 return false;
605 }
606
607 XFILEINI fileini;
608 XLICENSEID licenseID;
609 XSTRING licenseIDstring;
610 XSTRING licenseIDstring2;
611 XSTRING licensestring;
612 XSTRING licenseexprirationstring;
613 XBUFFER* _license = license;
614
615 if(!_license) _license = this->license;
616 if(!_license)
617 {
618 return false;

621 fileini.AddBufferLines(XFILETXTFORMATCHAR_ASCII, xbuffer);

623 fileini.ConvertFromLines();

625 fileini.ReadValue(XLICENSE_FILESECTION, XLICENSE_FILEVALUE_ID , licenseIDstring);

661bool XLICENSE::CheckMasterCreation(XPATH& xpath, XLICENSEID& licenseID, XSTRING& applicationID, int expirationdays)
662{
663 XFILEINI fileini;
664 XSTRING licensestring;
665
666 if(!fileini.Open(xpath))
667 {
668 return false;
669 }
670
671 fileini.ReadValue(XLICENSE_FILESECTION, XLICENSE_FILEVALUE_LICENSE, licensestring);
672 fileini.Close();
673
674 if(!applicationID.Compare(licensestring))
675 {
676 XFILEINI fileini;
677 XSTRING licenseIDstring;
678 XSTRING licensestring;
679 XSTRING licenseexpiration;
680 XBUFFER licencia;
681 XDATETIME* expirationdatetime;
682
683 licenseID.GetID()->GetToString(licenseIDstring);
684 if(!Generate(licenseID, &applicationID, &licencia))
685 {
686 return false;
687 }
688
689 licensestring.ConvertHexStringFromBuffer(licencia);
690
691 GEN_XFACTORY_CREATE(expirationdatetime, CreateDateTime())
692 if(!expirationdatetime)
693 {
694 return false;
695 }
696
697 expirationdatetime->SetToZero();
698 if(expirationdays)
699 {
700 expirationdatetime->Read();
701 expirationdatetime->AddDays((int)expirationdays);
702 }
703
704 expirationdatetime->GetDateTimeToString(XDATETIME_FORMAT_STANDARD, licenseexpiration);
705 GEN_XFACTORY.DeleteDateTime(expirationdatetime);
706
707 if(!CipherExpirationDate(true, applicationID, licenseexpiration))
708 {
709 return false;
710 }
711
712 if(!fileini.Create(xpath))
713 {
714 return false;
715 }
716
717 fileini.CreateSection(XLICENSE_FILESECTION);
718 fileini.WriteValue(XLICENSE_FILESECTION, XLICENSE_FILEVALUE_ID , licenseIDstring);
719 fileini.WriteValue(XLICENSE_FILESECTION, XLICENSE_FILEVALUE_LICENSE , licensestring);
720 fileini.WriteValue(XLICENSE_FILESECTION, XLICENSE_FILEVALUE_EXPIRATION, licenseexpiration);
721
722 fileini.Close();
723
724 return true;
725 }
726
727 return false;
728}
