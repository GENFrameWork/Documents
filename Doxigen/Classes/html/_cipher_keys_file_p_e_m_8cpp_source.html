240bool CIPHERKEYSFILEPEM::DecodeCertificates(XVECTOR<XSTRING*>* lines)
241{
242 XVECTOR<CIPHERKEYSFILEPEM_ENTRYBUFFER*> entrysbuffer;
243 CIPHERKEYSFILEPEM_ENTRYBUFFER* entrybuffer = NULL;
244
245 ndecodeobj = 0;
246
247 for(XDWORD c=0; c<lines->GetSize(); c++)
248 {
249 XSTRING line;
250 int ini = XSTRING_NOTFOUND;
251 int end = XSTRING_NOTFOUND;
252
253 if(lines->Get(c))
254 {
255 line = lines->Get(c)->Get();
256 }
257
258 if(line.IsEmpty())
259 {
260 continue;
261 }
262
263 ini = line.Find(CIPHERKEYSFILEPEM_BEGINKEY, true);
264 if(ini != XSTRING_NOTFOUND)
265 {
266 ini += XSTRING::GetSize(CIPHERKEYSFILEPEM_BEGINKEY);
267
268 if(!entrybuffer)
269 {
270 entrybuffer = new CIPHERKEYSFILEPEM_ENTRYBUFFER();
271 if(entrybuffer)
272 {
273 entrybuffer->end_line = XSTRING_NOTFOUND;
274 entrybuffer->ini_line = c+1;
275 end = line.Find(CIPHERKEYSFILEPEM_FINISH, true, ini);
276 if(end != XSTRING_NOTFOUND)
277 {
278 entrybuffer->type.AdjustSize(_MAXSTR);
279 line.Copy(ini, end, entrybuffer->type);
280 entrybuffer->type.AdjustSize();
281 }
282 else
283 {
284 delete entrybuffer;
285 entrybuffer = NULL;
286 }
287 }
288 }
289 }
290
291 ini = line.Find(CIPHERKEYSFILEPEM_ENDKEY, true);
292 if(ini != XSTRING_NOTFOUND)
293 {
294 ini += XSTRING::GetSize(CIPHERKEYSFILEPEM_ENDKEY);
295
296 if(entrybuffer)
297 {
298 entrybuffer->end_line = c;
299 end = line.Find(CIPHERKEYSFILEPEM_FINISH, true, ini);
300 if(end != XSTRING_NOTFOUND)
301 {
302 XSTRING type;
303
304 type.AdjustSize(_MAXSTR);
305 line.Copy(ini, end, type);
306 type.AdjustSize();
307
308 if(!type.Compare(entrybuffer->type))
309 {
310 entrysbuffer.Add(entrybuffer);
311 entrybuffer = NULL;
312 }
313 else
314 {
315 delete entrybuffer;
316 entrybuffer = NULL;
317 }
318 }
319 else
320 {
321 delete entrybuffer;
322 entrybuffer = NULL;
323 }
324 }
325 }
326 }
327
328
329 for(XDWORD c=0; c<entrysbuffer.GetSize(); c++)
330 {
331 CIPHERKEYSFILEPEM_ENTRYBUFFER* entrybuffer = entrysbuffer.Get(c);
332 if(entrybuffer)
333 {
334 XSTRING datastr;
335 XSTRING datastr1;
336
337 for(int d=entrybuffer->ini_line; d<entrybuffer->end_line; d++)
338 {
339 datastr += lines->Get(d)->Get();
340 }
341
342 if(!entrybuffer->data.ConvertFromBase64(datastr))
343 {
344 XFEEDBACK_ADD(XFEEDBACK_CODE_ERRORINFUNCTION , 1, __L("CIPHERKEYSFILEPEM::DecodeCertificates invalid Convert From Base64"));
345 return false;
346 }
347 }
348 }
349
350
351 XASN1 asn1;
352
353 for(XDWORD c=0; c<entrysbuffer.GetSize(); c++)
354 {
355 CIPHERKEYSFILEPEM_ENTRYBUFFER* entrybuffer = entrysbuffer.Get(c);
356 if(entrybuffer)
357 {
358 decodeobj = NULL;
359
360 if(!entrybuffer->type.Compare(__L("CERTIFICATE")))
361 {
362 decodeobjtype = CIPHERKEYSFILETYPE_CERTIFICATEX509;
363
364 decodeobj = (CIPHERKEY*)new CIPHERCERTIFICATEX509();
365 if(decodeobj)
366 {
367 if(asn1.Decode(entrybuffer->data, this))
368 {
369 Key_Add((CIPHERKEY*)decodeobj);
370
371 CIPHERCERTIFICATEX509* certificate = (CIPHERCERTIFICATEX509*)decodeobj;
372 if(certificate)
373 {
374 certificate->XTraceCertificatedPropertys();
375 }
376 }
377 else
378 {
379 delete (CIPHERCERTIFICATEX509*)decodeobj;
380 }
381 }
382 }
383 }
384 }
385
386
387 entrysbuffer.DeleteContents();
388 entrysbuffer.DeleteAll();
389
390 return true;
391}
