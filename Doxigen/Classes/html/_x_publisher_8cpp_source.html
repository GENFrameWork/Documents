460bool XPUBLISHER::PostEvent(XEVENT* event, XSUBJECT* subject)
461{
462 if(!on) return false;
463
464 inevent = true;
465
466 if(xmutexevents) xmutexevents->Lock();
467
468 XVECTOR<XOBSERVER*> observers;
469
470 for(XDWORD c=0; c<eventsvector.GetSize(); c++)
471 {
472 XPUBLISHERENTRY* publisherentry = eventsvector.Get(c);
473 if(publisherentry)
474 {
475 if((publisherentry->type == event->GetEventType()) && (publisherentry->subject == subject))
476 {
477 observers.Add(publisherentry->observer);
478 }
479 }
480 }
481
482 if(xmutexevents) xmutexevents->UnLock();
483
484 for(XDWORD c=0; c<observers.GetSize(); c++)
485 {
486 XOBSERVER* observer = observers.Get(c);
487 if(observer) observer->HandleEvent(event);
488 }
489
490 observers.DeleteAll();
491
492 inevent = false;
493
494 return true;
495}
