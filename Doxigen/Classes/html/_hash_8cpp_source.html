211bool HASH::Do(XPATH& xpath, XQWORD size, XQWORD pos)
212{
213 XFILE* xfile;

215
216 GEN_XFACTORY_CREATE(xfile, Create_File())

219 ResetResult();
220
221 if(xfile->Open(xpath, true))

251bool HASH::Do(XFILE* xfile, XQWORD size, XQWORD pos)
252{
253 if(!xfile) return false;
254 if(!xfile->IsOpen()) return false;
255
256 XBUFFER xbuffer;
257 XQWORD block = (((size==HASHALLFILESIZE)?xfile->GetSize():size) / HASHMAXFILESIZEBUFFER);
258 XDWORD mod = (((size==HASHALLFILESIZE)?xfile->GetSize():size) % HASHMAXFILESIZEBUFFER);
259 bool status = true;
260
261 if(!xfile->SetPosition(pos)) return false;
262
263 xbuffer.Resize((XDWORD)HASHMAXFILESIZEBUFFER);
264
265 for(XDWORD c=0;c<block;c++)
266 {
267 if(xfile->Read(xbuffer.Get(), HASHMAXFILESIZEBUFFER))
268 {
269 if(!Do(xbuffer))
270 {
271 status = false;
272 break;
273 }
274 }
275 else
276 {
277 status = false;
278 break;
279 }
280 }
281
282 if(status)
283 {
284 xbuffer.Resize(mod);
285 if(xfile->Read(xbuffer.Get(),mod))
286 {
287 if(!Do(xbuffer))
288 {
289 status = false;
290 }
291 }
292 else
293 {
294 status = false;
295 }
296 }
297
298 return status;
299}
