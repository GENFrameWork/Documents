541bool DIOPCAP_FILTERS::Entrys_Check(DIOPCAPFRAME* frame)
542{
543 bool valid = false;
544
545 if(!frame)
546 {
547 return false;
548 }
549
550 if(!entrys.GetSize())
551 {
552 return true;
553 }
554
555 for(XDWORD c=0; c<entrys.GetSize(); c++)
556 {
557 DIOPCAP_FILTER_ENTRY* entry = entrys.Get(c);
558 if(entry)
559 {
560 XDWORD protocoltype = entry->GetProtocolType();
561
562 if((protocoltype & DIOPCAPPROTOCOL_TYPE_ICMP) == frame->GetProtocolType())
563 {
564 valid = true;
565 }
566
567 if((protocoltype & DIOPCAPPROTOCOL_TYPE_UDP) == frame->GetProtocolType())
568 {
569 valid = true;
570 }
571
572 if((protocoltype & DIOPCAPPROTOCOL_TYPE_TCP) == frame->GetProtocolType())
573 {
574 valid = true;
575 }
576
577 if(valid)
578 {
579 if(!entry->Ports_GetAll()->GetSize())
580 {
581 return valid;
582 }
583
584 for(XDWORD d=0; d<entry->Ports_GetAll()->GetSize(); d++)
585 {
586 XDWORD port_target = entry->Ports_GetAll()->Get(d);
587 XDWORD port_source = (port_target & ~DIOPCAP_FILTERS_PORT_TARGET);
588
589 if(port_source == frame->GetSourcePort())
590 {
591 return valid;
592 }
593
594 if(port_target == frame->GetTargetPort())
595 {
596 return valid;
597 }
598 }
599 }
600 }
601 }
602
603 return false;
604}
