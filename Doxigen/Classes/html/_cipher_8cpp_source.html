365bool CIPHER::Cipher(XBYTE* input,XDWORD size)
366{
367 result->Delete();
368 result->Add(input, size);
369
370 CIPHERKEYSYMMETRICAL* key = (CIPHERKEYSYMMETRICAL*)GetKey(0);
371 if(!key) return false;
372 if(!key->Get()->Get()) return false;
373
374 int indexkey = 0;
375
376 for(XDWORD c=0; c<size; c++)
377 {
378 result->Get()[c] ^= key->Get()->GetByte(indexkey);
379
380 indexkey++;
381 if(indexkey >= (int)key->Get()->GetSize()) indexkey = 0;
382 }
383
384 return true;
385}
