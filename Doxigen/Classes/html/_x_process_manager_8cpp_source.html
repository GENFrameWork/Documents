772
773 target_buffer.Empty();

931bool XPROCESSMANAGER::Application_Terminate(XCHAR* processname, XDWORD exitcode)
932{
933 XVECTOR<XPROCESS*> apprunning;
934 bool status = false;
935
936 if(!GEN_XPROCESSMANAGER.Application_GetRunningList(apprunning)) return false;
937
938 for(XDWORD c=0; c<apprunning.GetSize(); c++)
939 {
940 XPROCESS* xprocess = apprunning.Get(c);
941 if(xprocess)
942 {
943 if(!xprocess->GetName()->Compare(processname, true))
944 {
945 status = Application_Terminate((*xprocess), exitcode);
946 break;
947 }
948 }
949 }
950
951 apprunning.DeleteContents();
952 apprunning.DeleteAll();
953
954 return status;
955}
