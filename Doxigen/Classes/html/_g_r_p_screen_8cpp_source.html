809bool GRPSCREEN::CreateViewport(XCHAR* ID, float posx, float posy, float width, float height, XDWORD canvasposx, XDWORD canvasposy, XDWORD canvaswidth, XDWORD canvasheight)
810{
811 GRPVIEWPORT* viewport = new GRPVIEWPORT();
812 if(!viewport) return false;
813
814 viewport->GetID()->Set(ID);
815 viewport->SetPosition(posx, posy);
816 viewport->SetSize(width, height);
817
818 GRPPROPERTIES canvasproperties;
819 float cwidth = (float)canvaswidth;
820 float cheight = (float)canvasheight;
821
822 if(width > cwidth) cwidth = width;
823 if(height > cheight) cheight = height;
824
825 canvasproperties.CopyPropertysFrom(this);
826 canvasproperties.SetPosition(canvasposx, canvasposy);
827 canvasproperties.SetSize((XDWORD)cwidth, (XDWORD)cheight);
828
829 if(viewport->CreateCanvas(canvasproperties))
830 {
831 viewport->SetCanvasPosition(posx, posy);
832 viewports.Add(viewport);
833
834 } else return false;
835
836 return true;
837 }

1051bool GRPSCREEN::Rotate(XBYTE* target, GRPCANVAS* screencanvas)
1052{
1053 XDWORD pixelsize = screencanvas->GetBytesperPixel();
1054 XBYTE* source = screencanvas->Buffer_Get();
1055
1056 if(!source || !target)
1057 {
1058 return false;
1059 }
1060
1061 switch(rotation)
1062 {
1063 case GRPSCREENROTATION_NONE : break;
1064
1065 case GRPSCREENROTATION_90_CLOCKWISE : { int target_offset = 0; //((x * height) + (height - y - 1)) * pixelsize;
1066 int source_offset = 0; //(y * width + x) * pixelsize;
1067
1068 for(XDWORD y=0; y<10 /*height*/; y++)
1069 {
1070 for(XDWORD x=0; x<width; x++)
1071 {
1072 //target_offset = (y * width + x) * pixelsize;
1073 //source_offset = (y * height + (width - x - 1)) * pixelsize;
1074
1075 target_offset += pixelsize;
1076 source_offset += pixelsize;
1077
1078 memcpy(target + target_offset, source + source_offset, pixelsize);
1079 }
1080 }
1081 }
1082 break;
1083
1084 case GRPSCREENROTATION_90_ANTICLOCKWISE :
1085 break;
1086
1087 case GRPSCREENROTATION_180 :
1088 break;
1089 }
1090
1091 return true;
1092}
