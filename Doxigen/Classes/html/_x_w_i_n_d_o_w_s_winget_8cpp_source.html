763bool XWINDOWSWINGET::Exec(XCHAR* params, XBUFFER& output)
764{
765 output.Delete();
766
767 // Construir comando PowerShell (igual que antes)
768 XSTRING cmd_str;
769 //cmd_str.Format(__L("powershell.exe -NonInteractive -Command \"%s\" 2>&1"), params);
770 cmd_str.Format(__L("pwsh.exe -NonInteractive -Command \"%s\" 2>&1"), params);
771
772
773 #ifdef SHOW_EXTRA_TRACE
774 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[WinGet] Command ejecuted: %s"), cmd_str.Get());
775 #endif
776
777 // Crear pipe para capturar salida
778 SECURITY_ATTRIBUTES sa;
779 STARTUPINFO si;
780 PROCESS_INFORMATION pi;
781 HANDLE hread = NULL;
782 HANDLE hwrite = NULL;
783
784 ZeroMemory(&si, sizeof(si));
785 ZeroMemory(&pi, sizeof(pi));
786
787 sa.nLength = sizeof(sa);
788 sa.lpSecurityDescriptor = NULL;
789 sa.bInheritHandle = TRUE;
790
791 if(!CreatePipe(&hread, &hwrite, &sa, 0))
792 {
793 return false;
794 }
795
796 // El extremo de lectura NO debe heredarse
797 SetHandleInformation(hread, HANDLE_FLAG_INHERIT, 0);
798
799 si.cb = sizeof(si);
800 si.dwFlags = STARTF_USESTDHANDLES | STARTF_USESHOWWINDOW;
801 si.wShowWindow = SW_HIDE;
802 si.hStdOutput = hwrite;
803 si.hStdError = hwrite;
804 si.hStdInput = GetStdHandle(STD_INPUT_HANDLE);
805
806 BOOL ok = CreateProcess( NULL,
807 (XCHAR*)cmd_str.Get(), // buffer modificable
808 NULL,
809 NULL,
810 TRUE, // heredar handles
811 CREATE_NO_WINDOW, // CLAVE: evita CMD
812 NULL,
813 NULL,
814 &si,
815 &pi
816 );
817
818 CloseHandle(hwrite);
819
820 if(!ok)
821 {
822 CloseHandle(hread);
823 return false;
824 }
825
826 BYTE buffer[4096];
827 DWORD bytes_read = 0;
828
829 while(ReadFile(hread, buffer, sizeof(buffer), &bytes_read, NULL) && bytes_read > 0)
830 {
831 output.Add(buffer, (XDWORD)bytes_read);
832 }
833
834 CloseHandle(hread);
835
836 WaitForSingleObject(pi.hProcess, INFINITE);
837
838 CloseHandle(pi.hThread);
839 CloseHandle(pi.hProcess);
840
841 NormalizeUnicode(output.Get());
842
843 #ifdef SHOW_EXTRA_TRACE
844 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[WinGet] Result: %s"), cmd_str.Get());
845 XTRACE_PRINTDATABLOCKCOLOR(XTRACE_COLOR_BLUE, output);
846 #endif
847
848
849 return true;
850}

885bool XWINDOWSWINGET::LooksLikeUTF8(XBYTE* buffer, int size)
886{
887 int i = 0;
888
889 while(i < size)
890 {
891 XBYTE c = buffer[i];
892
893 if(c < 0x80)
894 {
895 i++;
896 continue;
897 }
898
899 if((c & 0xE0) == 0xC0 && i + 1 < size && (buffer[i+1] & 0xC0) == 0x80)
900 {
901 i += 2;
902 continue;
903 }
904
905 if((c & 0xF0) == 0xE0 && i + 2 < size && (buffer[i+1] & 0xC0) == 0x80 && (buffer[i+2] & 0xC0) == 0x80)
906 {
907 i += 3;
908 continue;
909 }
910
911 return false;
912 }
913
914 return true;
915}

1069bool XWINDOWSWINGET::GenerateList(XSTRING& result, XVECTOR<XSTRING*>* list)
1070{
1071 XSTRING literal1 = __L("\r\n\x1b[32;1m--");
1072 XSTRING literal2 = __L("\x1B[0m\r\n");
1073 int start = XSTRING_NOTFOUND;
1074 int end = XSTRING_NOTFOUND;
1075
1076 start = result.Find(literal1.Get(), true);
1077 if(start == XSTRING_NOTFOUND)
1078 {
1079 return false;
1080 }
1081
1082 start += literal1.GetSize();
1083
1084 end = result.Find(literal2.Get(), true, start);
1085 if(end == XSTRING_NOTFOUND)
1086 {
1087 return false;
1088 }
1089
1090 start = end + literal2.GetSize();;
1091
1092 while(1)
1093 {
1094 end = result.Find(__L("\r\n"), true, start);
1095 if(end == XSTRING_NOTFOUND)
1096 {
1097 break;
1098 }
1099
1100 XSTRING* element = new XSTRING();
1101 if(!element)
1102 {
1103 break;
1104 }
1105
1106 result.Copy(start, end, (*element));
1107
1108 element->DeleteCharacter(__C(' '), XSTRINGCONTEXT_TO_END);
1109
1110 list->Add(element);
1111
1112 start = end + 2;
1113 }
1114
1115 if(!list->GetSize())
1116 {
1117 return false;
1118 }
1119
1120 return true;
