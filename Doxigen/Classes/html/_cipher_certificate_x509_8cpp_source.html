629bool CIPHERCERTIFICATEX509::IsValidDates()
630{
631 XDATETIME* xdatetime_notbefore = GetDateNotBefore();
632 XDATETIME* xdatetime_notafter = GetDateNotAfter();
633
634 if(!xdatetime_notbefore || !xdatetime_notafter)
635 {
636 return false;
637 }
638
639 XQWORD seconds_notbefore = xdatetime_notbefore->GetSeconsFromDate();
640 XQWORD seconds_notafter = xdatetime_notafter->GetSeconsFromDate();
641 XQWORD seconds_actual = 0;
642
643 XDATETIME* xdatetime = GEN_XFACTORY.CreateDateTime();
644 if(xdatetime)
645 {
646 xdatetime->Read();
647 seconds_actual = xdatetime->GetSeconsFromDate();
648
649 GEN_XFACTORY.DeleteDateTime(xdatetime);
650 }
651
652 if((seconds_actual > seconds_notbefore) && (seconds_actual < seconds_notafter))
653 {
654 return true;
655 }
656
657 return false;
658}

927bool CIPHERCERTIFICATEX509::ConvertDateTime(XCHAR* datestr, XDATETIME* datetime)
928{
929 if(!datestr)
930 {
931 return false;
932 }
933
934 if(!datetime)
935 {
936 return false;
937 }
938
939 XSTRING year;
940 XSTRING month;
941 XSTRING day;
942
943 XSTRING hour;
944 XSTRING minutes;
945 XSTRING seconds;
946
947 XSTRING datestring;
948
949 if(datestr[0] == __C('9'))
950 {
951 year = __L("19");
952 }
953 else
954 {
955 year = __L("20");
956 }
957
958 year += datestr[0];
959 year += datestr[1];
960
961 month += datestr[2];
962 month += datestr[3];
963
964 day += datestr[4];
965 day += datestr[5];
966
967 hour += datestr[6];
968 hour += datestr[7];
969
970 minutes += datestr[8];
971 minutes += datestr[9];
972
973 seconds += datestr[10];
974 seconds += datestr[11];
975
976 datestring.Format(__L("%s/%s/%s %s:%s:%s"), year.Get(), month.Get(), day.Get(), hour.Get(), minutes.Get(), seconds.Get());

978 datetime->GetDateTimeFromString(datestring, XDATETIME_FORMAT_POSTGRESQL);
