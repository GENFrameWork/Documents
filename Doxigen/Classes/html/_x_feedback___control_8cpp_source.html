311bool XFEEDBACK_CONTROL::AddFeedbak(char const* namefile, XFEEDBACK_CODE code,int line, ...)
312{
313 if(!isactive)
314 {
315 return false;
316 }
317
318 XFEEDBACK* feedback = new XFEEDBACK();
319 if(feedback)
320 {
321 feedback->SetCode(code);
322 feedback->GetModuleName()->Set(namefile);
323 feedback->SetModuleLine(line);
324
325 XSTRING text;
326
327 if(GetFeedbakText(code, text))
328 {
329 if(!text.IsEmpty())
330 {
331 va_list arg;
332
333 va_start(arg, line);
334
335 switch(code)
336 {
337 case XFEEDBACK_CODE_UNKNOWN : feedback->GetText()->Set(text);
338 break;
339
340 case XFEEDBACK_CODE_INVALIDPARAMFUNCTION : { int indexparam = va_arg(arg, int);
341 XCHAR* nameparam = va_arg(arg, XCHAR*);
342 XCHAR* namefunction = va_arg(arg, XCHAR*);
343
344 if(nameparam && namefunction)
345 {
346 feedback->GetText()->Format(text.Get(), indexparam, nameparam, namefunction);
347 }
348 }
349 break;
350
351 case XFEEDBACK_CODE_INVALIDMEMORYALLOCATION : { XCHAR* namevariable = va_arg(arg, XCHAR*);
352 XCHAR* namefunction = va_arg(arg, XCHAR*);
353
354 if(namevariable && namefunction)
355 {
356 feedback->GetText()->Format(text.Get(), namevariable, namefunction);
357 }
358 }
359 break;
360
361 case XFEEDBACK_CODE_ERRORINFUNCTION : { XCHAR* namefunction = va_arg(arg, XCHAR*);
362 XCHAR* errortext = va_arg(arg, XCHAR*);
363
364 if(namefunction && errortext)
365 {
366 feedback->GetText()->Format(text.Get(), namefunction, errortext);
367 }
368 }
369 break;
370
371 default : feedback->GetText()->Set(text);
372 break;
373 }
374
375 va_end(arg);
376
377
378
379 GEN_XLOG.AddEntry(XLOGLEVEL_WARNING, XFEEDBACK_CONTROL_LOG_SECTIONID, false , __L("%05d line %d, module %s -> %s"), feedbacks.GetSize() + 1
380 , feedback->GetModuleLine()
381 , feedback->GetModuleName()->Get()
382 , feedback->GetText()->Get());
383
384
385
386 return feedbacks.Add(feedback);
387 }
388 }
389 }
390
391 delete feedback;
392
393 return false;
394}
