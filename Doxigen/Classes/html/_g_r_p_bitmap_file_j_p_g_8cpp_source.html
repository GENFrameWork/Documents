347bool GRPBITMAPFILEJPG::ReadHeaderFile(GRPBITMAPFILEJPGHEADER* header)
348{
349 if(!header)
350 {
351 return false;
352 }
353
354 struct GRPBITMAPFILEJPG_ERROR jerr;
355
356 if(!GetPrimaryFile())
357 {
358 return false;
359 }
360
361 GetPrimaryFile()->SetPosition(0);
362
363 dinfo.err = jpeg_std_error(&jerr.pub);
364 jerr.pub.error_exit = GRPBITMAPFILEJPG_ErrorExit;
365
366 if(setjmp(jerr.setjmp_buffer))
367 {
368 jpeg_destroy_decompress(&dinfo);
369 Close();
370 return false;
371 }
372
373 jpeg_create_decompress(&dinfo); // Now we can initialize the JPEG decompression object.
374//jpeg_stdio_src(&dinfo, GetPrimaryFile()->GetFileStructHandle()); // Step 2: specify data source (eg, a file)
375
376
377 XQWORD file_size = GetPrimaryFile()->GetSize();
378 XBYTE* file_buffer = new XBYTE[(XDWORD)file_size];
379
380 GetPrimaryFile()->SetPosition(0);
381 GetPrimaryFile()->Read(file_buffer, (XDWORD)file_size);
382
383 jpeg_mem_src(&dinfo, file_buffer, (XDWORD)file_size);
384 jpeg_read_header(&dinfo, TRUE);
385
386 header->width = dinfo.image_width;
387 header->height = dinfo.image_height;
388 header->size = (header->height*header->width*3);
389
390 jpeg_destroy_decompress(&dinfo);
391
392 GetPrimaryFile()->SetPosition(0);
393
394 delete [] file_buffer;
395
396 return true;
397}

459{

488//jpeg_stdio_src(&dinfo, GetPrimaryFile()->GetFileStructHandle()); // Step 2: specify data source (eg, a file)
489
490
491 XQWORD file_size = GetPrimaryFile()->GetSize();
492 XBYTE* file_buffer=new XBYTE[(XDWORD)file_size];
493
494 GetPrimaryFile()->SetPosition(0);
495 GetPrimaryFile()->Read(file_buffer, (XDWORD)file_size);
496 jpeg_mem_src(&dinfo, file_buffer, (XDWORD)file_size);
497
498 jpeg_read_header(&dinfo, TRUE);
499
500 jpeg_start_decompress(&dinfo);
501
502 rowstride = dinfo.output_width * dinfo.output_components;
503
504 line = (*dinfo.mem->alloc_sarray)((j_common_ptr) &dinfo, JPOOL_IMAGE, rowstride, 1);
505
506 int size = dinfo.output_width * dinfo.output_height * dinfo.output_components;
507 int index = (size-rowstride);
508
509 memset(buffer, 0 ,size);
510
511 while (dinfo.output_scanline < dinfo.output_height)
512 {
513 jpeg_read_scanlines(&dinfo, line, 1);
514
515 memcpy((XBYTE*)&buffer[index],(XBYTE*)(*line),rowstride);
516 index-=rowstride;
517 }
518
519 jpeg_finish_decompress(&dinfo);
520
521 jpeg_destroy_decompress(&dinfo);
522
523 GetPrimaryFile()->SetPosition(0);

619bool GRPBITMAPFILEJPG::EncodeToFile(int width, int height, XBYTE* buffer, int quality, bool baseline)
620{
621 if(!buffer)
622 {
623 return false;
624 }
625
626 if(!GetPrimaryFile())
627 {
628 return false;
629 }
630
631 struct jpeg_error_mgr jerr;
632
633 JSAMPROW rowpointer[1]; // pointer to JSAMPLE row[s]
634 int rowstride; // physical row width in image buffer
635
636 GetPrimaryFile()->SetPosition(0);
637
638 cinfo.err = jpeg_std_error(&jerr);
639
640 jpeg_create_compress(&cinfo);
641
642 FILE* filehandle = GetPrimaryFile()->CreateStructHandle();
643 if(!filehandle)
644 {
645 return false;
646 }
647
648 jpeg_stdio_dest(&cinfo, filehandle);
649
650 cinfo.image_width = width; // image width and height, in pixels
651 cinfo.image_height = height;
652 cinfo.input_components = 3; // # of color components per pixel
653 cinfo.in_color_space = JCS_RGB; // colorspace of input image
654
655 jpeg_set_defaults(&cinfo);
656
657 jpeg_set_quality(&cinfo, quality, baseline);
658
659 jpeg_start_compress(&cinfo, TRUE);
660
661 rowstride = width * 3; // JSAMPLEs per row in image_buffer
662
663 int size = width * height * 3;
664
665 size -= rowstride;
666
667 while (cinfo.next_scanline < cinfo.image_height)
668 {
669 rowpointer[0] = &buffer[size-(cinfo.next_scanline * rowstride)];
670
671 jpeg_write_scanlines(&cinfo, rowpointer, 1);
672 }
673
674 jpeg_finish_compress(&cinfo);
675
676 jpeg_destroy_compress(&cinfo);
677
678 GetPrimaryFile()->SetPosition(0);
679
680 return true;
681}
