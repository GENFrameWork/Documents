458bool CIPHERKEYSFILEGKF::ExportToPEMFile(CIPHERKEY* key, XSTRING& publicPEM)
459{
460 if(!key) return false;
461
462 bool status = false;
463
464 switch(key->GetType())
465 {
466 case CIPHERKEYTYPE_RSA_PUBLIC : { XBER beroui;
467 XBER bernull;
468 XBER berbitstring;
469 XBER bermodule;
470 XBER berexponent;
471 XBER berseq1;
472 XBER berseq2;
473 XBER berseq3;
474 CIPHERKEYPUBLICRSA* publickey = (CIPHERKEYPUBLICRSA*)key;
475 XMPINTEGER module;
476 XMPINTEGER exponent;
477 XBUFFER xbuffermodule;
478 XBUFFER xbuffermodule2;
479 XBUFFER xbufferexponent;
480 XBUFFER xbufferpem;
481 XSTRING string;
482 XSTRING stringbase64;
483 XSTRING _publicPEM;
484
485 publickey->Get(module, exponent);
486
487 module.GetToXBuffer(xbuffermodule2 , module.GetSize());
488 exponent.GetToXBuffer(xbufferexponent , exponent.GetSize());
489
490
491 xbuffermodule.Add((XBYTE)0);
492 xbuffermodule.Add(xbuffermodule2);
493 bermodule.SetINTEGER(xbuffermodule);
494 berexponent.SetINTEGER(xbufferexponent);
495
496 berseq2.Sequence_AddTo(bermodule);
497 berseq2.Sequence_AddTo(berexponent);
498 berseq2.SetTagType(berseq2.GetTagType()|(XBYTE)(XBERTYPE_ISCONSTRUCTED));
499
500 xbufferpem.Add((XBYTE)0);
501 berseq2.GetDump(xbufferpem);
502 berbitstring.SetBITSTRING(xbufferpem);
503
504 beroui.SetOID(__L("1.2.840.113549.1.1.1"));
505 bernull.SetNULL();
506
507 berseq1.Sequence_AddTo(beroui);
508 berseq1.Sequence_AddTo(bernull);
509 berseq1.SetTagType(berseq1.GetTagType()|(XBYTE)(XBERTYPE_ISCONSTRUCTED));
510
511 berseq3.Sequence_AddTo(berseq1);
512 berseq3.Sequence_AddTo(berbitstring);
513 berseq3.SetTagType(berseq3.GetTagType()|(XBYTE)(XBERTYPE_ISCONSTRUCTED));
514
515 xbufferpem.Delete();
516 berseq3.GetDump(xbufferpem);
517
518 string.Set(xbufferpem.Get(), xbufferpem.GetSize());
519 string.ConvertToBase64(stringbase64);
520
521 _publicPEM.Empty();
522 _publicPEM += __L("-----BEGIN PUBLIC KEY-----\n");
523
524 int index = 0;
525 int lincounter = 0;
526
527 while(index<(int)stringbase64.GetSize())
528 {
529 _publicPEM.Add(stringbase64.Get()[index]);
530 index++;
531 lincounter++;
532
533 if(lincounter>=64)
534 {
535 _publicPEM.Add(__C('\n'));
536 lincounter = 0;
537 }
538 }
539
540 _publicPEM.Add(__C('\n'));
541 _publicPEM += __L("-----END PUBLIC KEY-----\n");
542
543 publicPEM = _publicPEM;
544 }
545 status = true;
546 break;
547
548 case CIPHERKEYTYPE_RSA_PRIVATE : break;
549
550 default : break;
551 }
552
553 return status;
554}

636bool CIPHERKEYSFILEGKF::ReadAllFile()

639
640 if(!xfilexml->Open(xpath)) return false;
641
642 xfilexml->ReadAndDecodeAllLines();

644 XFILEXMLELEMENT* noderoot = xfilexml->GetRoot();

648 {
649 XFILEXMLELEMENT* node = NULL;

651
652 do{ node = noderoot->GetElement(index);
653 if(node)
654 {

657 XSTRING string;

659 node->GetValueAttribute(CIPHERKEYSFILEGKF_ATTRNAME_CREATOR_CREATENAME , creatorname);

661 if(node->GetValueAttribute(CIPHERKEYSFILEGKF_ATTRNAME_CREATOR_LASTMODDATE , string)) creatorlastmodifieddatetime.GetDateTimeFromString(string, XDATETIME_FORMAT_STANDARD);
662 }

664 {

667 XSTRING string;

672 int typekey = string.ConvertToInt();

674 switch(typekey)

676 case CIPHERKEYTYPE_UNKNOWN : break;
