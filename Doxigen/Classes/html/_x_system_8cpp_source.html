735bool XSYSTEM::HostFile(bool add, XCHAR* domain, XCHAR* IP)
736{
737 XPATH xpath;
738 XFILETXT xfileTXT;
739 bool status = false;
740
741 #ifdef WINDOWS
742 xpath = __L("C:\\Windows\\System32\\drivers\\etc\\hosts");
743 #endif
744
745 #if defined(LINUX) || defined(ANDROID)
746 xpath = __L("/etc/hosts");
747 #endif
748
749 if(xfileTXT.Open(xpath, false))
750 {
751 xfileTXT.ReadAllFile();
752
753 if(add)
754 {
755 int nline = IsLineInHostFile(xfileTXT, domain, IP);
756 if(nline == XSTRING_NOTFOUND)
757 {
758 XSTRING line;
759
760 line.Format(__L("%s %s"), IP, domain);
761
762 xfileTXT.AddLine(line);
763 xfileTXT.WriteAllFile();
764 }
765 }
766 else
767 {
768 int nline = IsLineInHostFile(xfileTXT, domain, IP);
769 if(nline != XSTRING_NOTFOUND)
770 {
771 xfileTXT.DeleteLine(nline);
772 status = xfileTXT.WriteAllFile();
773 }
774 }
775
776 xfileTXT.Close();
777 }
778
779 return status;
780}

815int XSYSTEM::IsLineInHostFile(XFILETXT& xfileTXT, XCHAR* domain, XCHAR* IP)
816{
817 bool found = false;
818 int c = 0;
819
820 if(!xfileTXT.GetNLines()) return XSTRING_NOTFOUND;
821
822 do{ XSTRING* line = xfileTXT.GetLine(c);
823 if(line)
824 {
825 int index = line->Find(domain, true);
826 if(index != XSTRING_NOTFOUND)
827 {
828 found = true;
829
830 for(XDWORD d=0; d<line->GetSize(); d++)
831 {
832 XCHAR character = line->Get()[d];
833
834 if((character != __C('\t') && (character != __C(' '))))
835 {
836 if(character == __C('#'))
837 {
838 found = false;
839 }
840
841 break;
842 }
843 }
844
845 if(found) break;
846 }
847 }
848
849 c++;
850
851 } while(c<xfileTXT.GetNLines());
852
853 if(!found) return XSTRING_NOTFOUND;
854
855 return c;
856}
