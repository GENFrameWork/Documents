394bool XCONSOLE::PrintDataBlock(XBYTE* data, XDWORD size, XDWORD marginsize, XDWORD sizeline, bool showoffset, bool showtext)
395{
396 XSTRING margin;
397 XSTRING strdata;
398 XDWORD _size = 0;
399 XDWORD _sizeline = sizeline;
400 int index = 0;
401
402 for(XDWORD c=0;c<marginsize;c++)
403 {
404 margin += __L(" ");
405 }
406
407 while(_size<size)
408 {
409 XSTRING string;
410
411 string = margin;
412 if(_sizeline > (size - _size)) _sizeline = (size - _size);
413
414 if(showoffset)
415 {
416 strdata.Format(__L("%04X"),index);
417 string += strdata;
418 string += __L(" ");
419 }
420
421 for(XDWORD c=0; c<_sizeline; c++)
422 {
423 strdata.Format(__L("%02X "),data[index]);
424 string += strdata;
425 _size++;
426 index++;
427 }
428
429 if(_sizeline != sizeline)
430 {
431 for(XDWORD c=0; c<(sizeline-_sizeline); c++)
432 {
433 strdata.Format(__L(" "));
434 string += strdata;
435 }
436 }
437
438 if(showtext)
439 {
440 index -= _sizeline;
441 string += __L(" ");
442
443 for(XDWORD c=0; c<_sizeline; c++)
444 {
445 XCHAR character = (XCHAR)data[index];
446
447 if((character<0x20) || (character>0x80) || (character==__C('%')) || (character==__C('\\'))) character = __C('.');
448
449 strdata.Format(__L("%c"),character);
450 string += strdata;
451 index++;
452 }
453 }
454
455 Print(string.Get());
456 Print(__L("\n"));
457
458 if(_size > XCONSOLE_MAXSIZEDATABLOCK) break;
459 }
460
461 return true;
462}

558bool XCONSOLE::WaitKey(XCHAR* text, XDWORD margin, bool prelude, XDWORD timeout)
559{
560 XTIMER* GEN_XFACTORY_CREATE(xtimer, CreateTimer())
561 if(!xtimer) return false;
562
563 xtimer->Reset();
564
565 XSTRING string;
566 XSTRING string2;
567 bool status = true;
568
569
570 while(KBHit())
571 {
572 GetChar();
573 }
574
575 while(!KBHit())
576 {
577 string = text;
578 string += __L(" \r");
579
580 string2.Format(string.Get(), (timeout - xtimer->GetMeasureSeconds()));
581
582 GEN_XSLEEP.MilliSeconds(100);
583
584 PrintMessage(string2.Get(), margin, prelude, false);
585
586 if(timeout)
587 {
588 if((XDWORD)xtimer->GetMeasureSeconds() > timeout)
589 {
590 status = false;
591 break;
592 }
593 }
594 }
595
596 while(KBHit())
597 {
598 GetChar();
599 }
600
601 string.Empty();
602
603 for(int c=0; c<78; c++)
604 {
605 string += __L(" ");
606 }
607
608 string += __L("\r");
609 PrintMessage(string.Get(), 0, false, false);
610
611 GEN_XFACTORY.DeleteTimer(xtimer);
612
613 return status;
614}
