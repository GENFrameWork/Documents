291bool DIOCOREPROTOCOL_REGISTERDATA::CreateIDMachine(XUUID& ID)
292{
293 XSTRING origin;
294 XBUFFER originbuffer;
295 HASHSHA2* sha2 = NULL;
296 XBUFFER* result = NULL;
297
298 ID.Empty();
299
300 origin.AddFormat(__L(" %s"), GEN_XSYSTEM.GetBIOSSerialNumber()->Get());
301 origin.AddFormat(__L(" %s"), GEN_XSYSTEM.GetCPUSerialNumber()->Get());
302
303 origin.ConvertToUTF8(originbuffer);
304
305 sha2 = new HASHSHA2(HASHSHA2TYPE_256);
306 if(!sha2)
307 {
308 return false;
309 }
310
311 sha2->Do(originbuffer);
312 result = sha2->GetResult();
313
314 if(!result)
315 {
316 return false;
317 }
318
319 if(!result->GetSize())
320 {
321 return false;
322 }
323
324 XDWORD data1 = 0;
325 XWORD data2 = 0;
326 XWORD data3 = 0;
327 XBYTE data4 = 0;
328 XBYTE data5 = 0;
329 XBYTE* data6 = &result->Get()[12];
330
331 result->Get((XDWORD&)data1);
332 result->Get((XWORD&)data2);
333 result->Get((XWORD&)data3);
334 result->Get((XBYTE&)data4);
335 result->Get((XBYTE&)data5);
336
337 ID.Set(data1, data2, data3, data4, data5, data6);
338
339 ID.GetToString(origin);
340
341 delete sha2;
342
343 return true;
344}
