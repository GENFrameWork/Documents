891void XTHREADSCOLLECTED::ThreadRunFunction(void* param)
892{
893 XTHREADSCOLLECTED* xthreadscollected = (XTHREADSCOLLECTED*)param;
894 if(!xthreadscollected) return;
895
896 if(!xthreadscollected->xthreadsvectormutex) return;
897
898 xthreadscollected->xthreadsvectormutex->Lock();
899
900 for(XDWORD c=0; c<xthreadscollected->xthreadsvector.GetSize(); c++)
901 {
902 XTHREADCOLLECTED* xthreadcollected = xthreadscollected->xthreadsvector.Get(c);
903 if(xthreadcollected)
904 {
905 switch(xthreadcollected->GetStatus())
906 {
907 case XTHREADSTATUS_RUN : xthreadcollected->SetIsInFunction(true);
908 xthreadcollected->GetFunction()(xthreadcollected->GetParam());
909 xthreadcollected->SetIsInFunction(false);
910
911 xthreadscollected->xthread->Wait();
912 break;
913
914 case XTHREADSTATUS_EXIT : xthreadcollected->SetStatus(XTHREADSTATUS_END);
915 break;
916
917 default : break;
918 }
919 }
920 }
921
922 xthreadscollected->xthreadsvectormutex->UnLock();
923}
