283XFILETXTFORMATCHAR XFILETXT::GetFormatCharFromFile(XDWORD* sizeBOM)
284{
285 XFILETXTFORMATCHAR formatchar;
286
287 XBYTE BOM[3];
288 XQWORD position = 0;
289
290 file->GetPosition(position);
291 file->SetPosition(0);
292
293 if(sizeBOM) (*sizeBOM) = 0;
294
295 if(file->Read((XBYTE*)&BOM,3))
296 {
297 if((BOM[0]==0xEF)&&(BOM[1]==0xBB)&&(BOM[2]==0xBF))
298 {
299 formatchar = XFILETXTFORMATCHAR_UTF8;
300 if(sizeBOM) (*sizeBOM) = 3;
301 }
302 else
303 {
304 if((BOM[0]==0xFE)&&(BOM[1]==0xFF))
305 {
306 formatchar = XFILETXTFORMATCHAR_UTF16_BE;
307 if(sizeBOM) (*sizeBOM) = 2;
308 }
309 else
310 {
311 if((BOM[0]==0xFF)&&(BOM[1]==0xFE))
312 {
313 formatchar = XFILETXTFORMATCHAR_UTF16_LE;
314 if(sizeBOM) (*sizeBOM) = 2;
315 }
316 else formatchar = XFILETXTFORMATCHAR_ASCII;
317 }
318 }
319
320 } else formatchar = XFILETXTFORMATCHAR_ASCII;
321
322 file->SetPosition(position);
323
324 return formatchar;
325}

646bool XFILETXT::IsBinaryFile()
647{
648 if(!IsOpen()) return false;
649 if(!file) return false;
650
651 bool istextfile = false;
652
653 #define MAX_BUFFER_TEST 1024*5
654
655 XBUFFER buffer;
656 XDWORD size = MAX_BUFFER_TEST;
657 XDWORD controlchars = 0;
658
659 buffer.Resize(MAX_BUFFER_TEST);
660 file->Read(buffer.Get(), &size);
661
662 if(size)
663 {
664 for(XDWORD c=0; c<size; c++)
665 {
666 if((((int)buffer.Get()[c] >= 0x00) && ((int)buffer.Get()[c] < 0x09)) ||
667 (((int)buffer.Get()[c] > 0x0D) && ((int)buffer.Get()[c] < 0x20)))
668 {
669 controlchars++;
670 }
671 }
672
673 int percent = (controlchars*100/size);
674 if(percent < 10) istextfile = true;
675 }
676
677 return !istextfile;
678}

852
853 if(lines.IsEmpty()) return false;
854

880 for(int c=0;c<(int)lines.GetSize();c++)
881 {

886
887 if(string)
888 {
889 switch(formatchar)
890 {
891 case XFILETXTFORMATCHAR_UNKNOWN : break;
892
893 case XFILETXTFORMATCHAR_ASCII : { bw = string->GetSize();
894
895 XBUFFER charstr;
896
897 (*string).ConvertToASCII(charstr);
898 memcpy(buffer,(XBYTE*)charstr.Get(), bw);
899
900 }
901 break;

924 bufferw[d] = (XWORD)string->Get()[d];
925 }

930 }
